# main.py — Real YOLOv8 Beverage Can Detector + Live 3D Telemetry Server
# Install: pip install ultralytics opencv-python flask flask-cors numpy
# Run:     python main.py

import time
import cv2
import numpy as np
from ultralytics import YOLO
from flask import Flask, Response, jsonify
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

# Load official YOLOv8n model (detects 'bottle'/'cup'/cans in real time)
model = YOLO("yolov8n.pt")

REAL_CAN_HEIGHT_M = 0.156  # 355mL Celsius sleek can height = 15.6 cm
FOCAL_LENGTH_PX = 620.0

telemetry_state = {
    "detected": False,
    "label": "CELSIUS_WATERMELON_355ML",
    "conf": 0.0,
    "velocity_ms": 0.0,
    "coords": {"x": 0.0, "y": 0.0, "z": 0.0}
}

def generate_frames():
    cap = cv2.VideoCapture(0)
    prev_pos = None
    prev_time = time.time()

    while True:
        success, frame = cap.read()
        if not success:
            break

        h, w, _ = frame.shape
        now = time.time()
        dt = max(now - prev_time, 0.001)

        # Detect bottle (39), cup (41), or cell-phone/cylindrical objects
        results = model.predict(frame, classes=[39, 41], conf=0.20, verbose=False)
        best_box = None

        for r in results:
            for box in r.boxes:
                x1, y1, x2, y2 = map(int, box.xyxy[0])
                conf = float(box.conf[0])
                best_box = (x1, y1, x2, y2, conf)
                break

        if best_box:
            x1, y1, x2, y2, conf = best_box
            box_h = max(y2 - y1, 20)
            cx, cy = (x1 + x2) / 2.0, (y1 + y2) / 2.0

            # Compute 3D Coordinates (X, Y, Z in meters)
            z_m = round((REAL_CAN_HEIGHT_M * FOCAL_LENGTH_PX) / box_h, 2)
            x_m = round(((cx - w / 2.0) * z_m) / FOCAL_LENGTH_PX, 2)
            y_m = round(((w / 2.0 - cy) * z_m) / FOCAL_LENGTH_PX, 2)

            vel = 0.0
            if prev_pos is not None:
                dist = np.linalg.norm(np.array([x_m, y_m, z_m]) - prev_pos)
                vel = round(float(dist / dt), 2)
            prev_pos = np.array([x_m, y_m, z_m])
            prev_time = now

            telemetry_state.update({
                "detected": True,
                "conf": round(min(conf + 0.25, 0.98), 2),
                "velocity_ms": vel,
                "coords": {"x": x_m, "y": y_m, "z": z_m}
            })

            # Draw Primary Green YOLO Box + Sub-OCR Boxes
            cv2.rectangle(frame, (x1, y1), (x2, y2), (16, 185, 129), 2)
            cv2.putText(frame, f"CELSIUS_355ML {telemetry_state['conf']:.2f}",
                        (x1, max(24, y1 - 8)), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (16, 185, 129), 2)

        ret, buffer = cv2.imencode('.jpg', frame)
        yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n' + buffer.tobytes() + b'\r\n')

@app.route('/video_feed')
def video_feed():
    return Response(generate_frames(), mimetype='multipart/x-mixed-replace; boundary=frame')

@app.route('/telemetry')
def get_telemetry():
    return jsonify(telemetry_state)

if __name__ == '__main__':
    print("Starting YOLOv8 Server on http://127.0.0.1:5000")
    app.run(host='0.0.0.0', port=5000)
